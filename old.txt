class KeywordExtractor:
    def __init__(self):
        self.nlp = self._load_spacy()

    @lru_cache(maxsize=1)
    def _load_spacy(self):
        return spacy.load("en_core_web_md")

    def extract_keywords(self, text: str, max_keywords: int = 15) -> List[str]:
        doc = self.nlp(text)
        keywords = [
            token.lemma_.lower() for token in doc
            if token.pos_ in {'NOUN', 'PROPN', 'ADJ'}
            and not token.is_stop and token.is_alpha
        ]
        ner_keywords = [ent.text.lower() for ent in doc.ents
                        if ent.label_ in {'PERSON', 'ORG', 'GPE', 'PRODUCT', 'EVENT'}]
        combined = keywords + ner_keywords
        ranked = Counter(combined).most_common(max_keywords)
        return [kw for kw, _ in ranked]

class TFIDFQueryProcessor:
    def __init__(self, corpus: Optional[List[str]] = None):
        self.vectorizer = TfidfVectorizer(
            max_features=1000,
            stop_words='english',
            ngram_range=(1, 2),
            min_df=1,
            max_df=0.95
        )
        default_corpus = corpus or [
            "machine learning artificial intelligence",
            "web development programming",
            "data science analytics",
            "business strategy marketing",
            "technology innovation research"
        ]
        self.vectorizer.fit(default_corpus)

    def get_important_terms(self, query: str, top_n: int = 8) -> List[Tuple[str, float]]:
        query_vector = self.vectorizer.transform([query])
        feature_names = self.vectorizer.get_feature_names_out()
        scores = query_vector.toarray()[0]
        top_indices = np.argsort(scores)[-top_n:][::-1]
        return [(feature_names[i], scores[i]) for i in top_indices if scores[i] > 0]

class POSQueryAnalyzer:
    def __init__(self):
        self.nlp = spacy.load("en_core_web_md")

    def analyze_query_structure(self, query: str) -> Dict:
        doc = self.nlp(query)
        subj, verb = None, None
        for token in doc:
            if "subj" in token.dep_:
                subj = token.text
            elif token.dep_ == "ROOT":
                verb = token.text
        return {
            'main_subject': subj,
            'main_action': verb,
            'query_type': self._classify_query_type(query),
            'complexity_score': len(doc) / 10.0
        }

    def _classify_query_type(self, query: str) -> str:
        qwords = ['what', 'who', 'where', 'when', 'why', 'how']
        tokens = word_tokenize(query.lower())
        if any(q in tokens for q in qwords):
            return 'question'
        elif any(x in query.lower() for x in ['compare', 'vs', 'versus']):
            return 'comparative'
        elif any(pos.startswith('VB') for _, pos in pos_tag(tokens)):
            return 'action'
        else:
            return 'informational'

class KeywordRefiner:
    def __init__(self):
        self.stop_words = set(stopwords.words('english'))
        self.stop_words.update([
            'please', 'could', 'would', 'should', 'might', 'may',
            'find', 'search', 'look', 'show', 'tell', 'give', 'get',
            'want', 'need', 'help', 'information', 'about', 'regarding'
        ])
        self.lemmatizer = WordNetLemmatizer()
        self.punctuation_table = str.maketrans('', '', string.punctuation)

    def refine_keywords(self, text: str) -> List[str]:
        tokens = word_tokenize(text.lower())
        filtered = [
            self.lemmatizer.lemmatize(token.translate(self.punctuation_table))
            for token in tokens
            if token.isalpha() and token not in self.stop_words and len(token) > 2
        ]
        return list(dict.fromkeys(filtered))

class IntentRefiner:
    def __init__(self):
        self.nlp = spacy.load("en_core_web_md")
        self.temporal_keywords = {
            'recent': 30, 'latest': 7, 'current': 1, 'today': 0,
            'this week': 7, 'this month': 30, 'this year': 365,
            'yesterday': 1, 'last week': 7, 'last month': 30
        }

    def expand_synonyms(self, keywords: List[str], max_synonyms: int = 2) -> List[str]:
        doc = self.nlp(" ".join(keywords))
        expanded = set(keywords)
        for token in doc:
            if token.has_vector:
                similar = sorted(self.nlp.vocab, key=lambda w: token.similarity(w), reverse=True)
                for sim in similar[:max_synonyms]:
                    if sim.text not in expanded and sim.is_alpha and not sim.is_stop:
                        expanded.add(sim.text)
        return list(expanded)

    def add_temporal_context(self, query: str) -> str:
        q = query.lower()
        for term, days in self.temporal_keywords.items():
            if term in q:
                date_str = (datetime.now() - timedelta(days=days)).strftime("%Y-%m-%d")
                return f"{query} after:{date_str}"
        return query

    def refine_for_search_intent(self, original_query: str, keywords: List[str]) -> Dict:
        expanded_keywords = self.expand_synonyms(keywords)
        temporal_query = self.add_temporal_context(original_query)
        return {
            'original': original_query,
            'expanded_keywords': expanded_keywords,
            'alternative_queries': [
                ' '.join(expanded_keywords),
                f"({' OR '.join(keywords[:4])})",
                temporal_query,
                f'"{keywords[0]}" {" ".join(keywords[1:3])}' if len(keywords) > 2 else ' '.join(keywords)
            ],
            'recommended_query': ' '.join(expanded_keywords)
        }

class QueryStructurer:
    def __init__(self):
        self.platform_optimizations = {
            'google': {'optimal_length': (2, 10)},
            'bing': {'optimal_length': (3, 12)},
            'academic': {'optimal_length': (5, 15)},
            'duckduckgo': {'optimal_length': (2, 8)}
        }

    def structure_query(self, keywords: List[str], query_type: str = 'informational', platform: str = 'google') -> str:
        if not keywords:
            return ""
        if query_type == 'question':
            query = f"{keywords[0]} {' '.join(keywords[1:4])}"
        elif query_type == 'comparative' and len(keywords) >= 2:
            query = f'"{keywords[0]}" vs "{keywords[1]}" {' '.join(keywords[2:4])}'
        elif query_type == 'academic':
            query = f'{' '.join(keywords[:3])} research OR study OR analysis'
        else:
            query = f'"{keywords[0]}" {' '.join(keywords[1:4])}'

        return self._optimize_length(query, platform)

    def _optimize_length(self, query: str, platform_cfg: Dict) -> str:
        words = query.split()
        platform_settings = self.platform_optimizations.get(platform_cfg, self.platform_optimizations['google'])
        min_w, max_w = platform_settings['optimal_length']

        return ' '.join(words[:max_w])

    def generate_query_variations(self, keywords: List[str], query_type: str) -> Dict[str, str]:
        return {platform: self.structure_query(keywords, query_type, platform) for platform in self.platform_optimizations}

class WebSearchPipeline:
    def __init__(self):
        logger.info("Initializing Web Search Pipeline...")
        self.keyword_extractor = KeywordExtractor()
        self.tfidf_processor = TFIDFQueryProcessor()
        self.pos_analyzer = POSQueryAnalyzer()
        self.keyword_refiner = KeywordRefiner()
        self.intent_refiner = IntentRefiner()
        self.query_structurer = QueryStructurer()

    def process_query(self, user_query: str, platform: str = 'google', include_metadata: bool = True) -> Dict:
        start = time.time()
        initial_keywords = self.keyword_extractor.extract_keywords(user_query)
        tfidf_terms = self.tfidf_processor.get_important_terms(user_query)
        pos_data = self.pos_analyzer.analyze_query_structure(user_query)
        refined = self.keyword_refiner.refine_keywords(user_query)
        intent = self.intent_refiner.refine_for_search_intent(user_query, refined)
        final = self.query_structurer.structure_query(refined, pos_data['query_type'], platform)
        variations = self.query_structurer.generate_query_variations(refined, pos_data['query_type'])
        elapsed = time.time() - start

        result = {
            'original_query': user_query,
            'processed_query': final,
            'primary_keywords': refined[:5],
            'query_type': pos_data['query_type'],
            'platform_variations': variations,
            'processing_time_ms': round(elapsed * 1000, 2),
            'recommended_queries': intent['alternative_queries']
        }

        if include_metadata:
            result['metadata'] = {
                'initial_keywords': initial_keywords,
                'tfidf_terms': tfidf_terms,
                'pos_analysis': pos_data,
                'expanded_keywords': intent['expanded_keywords']
            }

        return result
